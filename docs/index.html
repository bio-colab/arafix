💻 التكامل البرمجي في مشاريعك (Python & CLI) 
# تثبيت النواة النقية (بدون أي تبعيات خارجية - stdlib only):
pip install arafix

import arafix

# إصلاح أي نص معطوب فورياً:
broken = "ةيبرعلا ةغللا 123"
clean = arafix.fix(broken)
print(clean)
# → 123 اللغة العربية
# تثبيت مع دعم PDF:
pip install "arafix[pdf]"

import arafix

# 1. استخراج نص كامل سليم بسطر واحد:
text = arafix.read("thesis.pdf")

# 2. استخراج مستند Markdown مهيكل مع العناوين والجداول:
md = arafix.read_markdown("thesis.pdf")

# 3. استخراج سياق فائق النقاء لـ LLMs بتوفير توكنز حتى 45%:
llm_ctx = arafix.read_llm("thesis.pdf", strip_tashkeel=True)
# استخراج ملف PDF مباشرة من الطرفية:
arafix document.pdf -o out.txt

# استخراج كـ Markdown مع الجداول:
arafix extract report.pdf --format markdown --tables -o report.md

# تشخيص العيوب الجنائية لملف PDF:
arafix diagnose thesis.pdf -v
