# تثبيت مع دعم PDF:
pip install "arafix[pdf]"

import arafix

# 1. استخراج فوري نقي
text = arafix.read("document.pdf")

# 2. استخراج Markdown مهيكل للذكاء الاصطناعي مع الجداول
md = arafix.read_markdown("document.pdf")

# 3. استخراج RAG مكاني جاهز للربط مع LangChain / LlamaIndex
chunks = arafix.extract_pdf_rag("document.pdf")
for chunk in chunks["chunks"]:
    print(chunk["text"], chunk["spatial"]["page"])
# استخراج ملف PDF مباشرة من الطرفية:
arafix document.pdf -o out.txt

# استخراج مع تصدير Markdown والجداول:
arafix extract report.pdf --format markdown --tables

# تشخيص العيوب الجنائية لملف PDF:
arafix diagnose thesis.pdf -v
